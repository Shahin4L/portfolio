function nameKey_(s) { return String(s || '').toLowerCase().replace(/\s+/g, ' ').trim(); }

/* ================= ACCESS ================= */
function getAccess_(email) {
  const em = normEmail_(email);
  const none = { authorized:false, email:em, name:'', role:'', salaryEdit:false, salaryView:false, orderCostView:false, configEdit:false, bankingView:false, status:false };
  if (!em) return none;
  const sh = audit_().getSheetByName(APP.ACCESS_SHEET);
  if (!sh) throw new Error('Access Control sheet not found. Please run setupApp() first.');
  if (sh.getLastRow() < 2) return none;
  const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 6).getValues();
  for (let i = 0; i < rows.length; i++) {
    if (normEmail_(rows[i][0]) !== em) continue;
    const ok = toBool_(rows[i][5]), role = String(rows[i][2] || ''), roleKey = role.toLowerCase().trim(), admin = roleKey === 'admin';
    const lead = roleKey === 'lead';
    const sEdit = toBool_(rows[i][3]);
    return { authorized: ok, email: em, name: String(rows[i][1] || ''), role: role,
      salaryEdit: ok && sEdit && !lead, salaryView: ok && (sEdit || admin) && !lead,
      orderCostView: ok && (lead || sEdit || admin),
      configEdit: ok && toBool_(rows[i][4]) && !lead, bankingView: ok && admin, status: ok };
  }
  return none;
}
// Resolves a session token to the signed-in email (CacheService). Throws SESSION_EXPIRED when missing/expired.
function session_(token) {
  const t = String(token || '');
  return t ? (CacheService.getScriptCache().get('sess_' + t) || '') : '';
}
function require_(token, perm) {
  const em = session_(token);
  if (!em) throw new Error('SESSION_EXPIRED: Your session has expired. Please sign in again.');
  const a = getAccess_(em);
  if (!a.authorized) throw new Error('Access denied. This email is not authorized.');
  if (perm && !a[perm]) throw new Error('You do not have permission for this action.');
  return a;
}

const OTP_SALT = 'csor-tracker-v1';                                          // any private string

function otpHash_(email, code) {
  const raw = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, normEmail_(email) + '|' + String(code) + '|' + OTP_SALT);
  return Utilities.base64Encode(raw);
}

// Step 1: send a one-time code to the user's email
function sendLoginCode(email) {
  const a = getAccess_(email);
  if (!a.authorized) {
    log_(email, 'LOGIN_DENIED', 'Unauthorized login attempt');
    return { success:false, message:'Access denied. This email is not authorized.' };
  }
  const cache = CacheService.getScriptCache();
  const rlKey = 'otp_rl_' + a.email;
  if (cache.get(rlKey)) return { success:false, message:'Please wait ' + OTP.RESEND_WAIT + ' seconds before requesting a new code.' };

  const code = String(Math.floor(100000 + Math.random() * 900000));
  cache.put('otp_' + a.email, JSON.stringify({ h:otpHash_(a.email, code), tries:0 }), OTP.TTL);
  cache.put(rlKey, '1', OTP.RESEND_WAIT);

  try {
    MailApp.sendEmail({
      to: a.email,
      name: 'Central Sort Ops Resource Tracker',
      subject: 'Your verification code: ' + code,
      body: 'Your verification code is ' + code + '\nIt is valid for ' + (OTP.TTL / 60) + ' minutes. Do not share this code with anyone.',
      htmlBody: otpEmailHtml_(code)
    });
  } catch (e) {
    cache.remove('otp_' + a.email); cache.remove(rlKey);
    log_(a.email, 'LOGIN_CODE_FAILED', e.message);
    return { success:false, message:'Unable to send the verification email: ' + e.message };
  }
  log_(a.email, 'LOGIN_CODE_SENT', 'Verification code sent');
  return { success:true, resendWait:OTP.RESEND_WAIT, message:'Verification code sent to ' + a.email };
}

// Step 2: verify the code and start a session
function verifyLoginCode(email, code) {
  const a = getAccess_(email);
  if (!a.authorized) return { success:false, authorized:false, message:'Access denied. This email is not authorized.' };
  code = String(code || '').replace(/\D/g, '');
  if (code.length !== 6) return { success:false, message:'Please enter the 6-digit code.' };

  const cache = CacheService.getScriptCache(), k = 'otp_' + a.email;
  const raw = cache.get(k);
  if (!raw) return { success:false, message:'Code expired or not requested. Please request a new code.' };
  const rec = JSON.parse(raw);

  if (rec.tries >= OTP.MAX_TRIES) {
    cache.remove(k);
    log_(a.email, 'LOGIN_CODE_LOCKED', 'Too many wrong attempts');
    return { success:false, message:'Too many wrong attempts. Please request a new code.' };
  }
  if (rec.h !== otpHash_(a.email, code)) {
    rec.tries++;
    cache.put(k, JSON.stringify(rec), OTP.TTL);
    log_(a.email, 'LOGIN_CODE_WRONG', 'Wrong code (' + rec.tries + '/' + OTP.MAX_TRIES + ')');
    return { success:false, message:'Incorrect code. ' + (OTP.MAX_TRIES - rec.tries) + ' attempt(s) left.' };
  }

  cache.remove(k);                                   // one-time use
  const token = Utilities.getUuid() + Utilities.getUuid();
  cache.put('sess_' + token, a.email, OTP.SESSION_TTL);   // token -> email
  log_(a.email, 'LOGIN', 'User logged in (verified)');
  return { success:true, authorized:true, user:a, token:token, bootstrap:bootstrap_(a) };
}

function logout(token) {
  const em = session_(token);
  CacheService.getScriptCache().remove('sess_' + String(token || ''));
  if (em) log_(em, 'LOGOUT', 'User logged out');
  return { success:true };
}
function getBootstrap(token) {
  const em = session_(token);
  if (!em) return { authorized:false };
  const a = getAccess_(em);
  if (!a.authorized) return { authorized:false };
  return { authorized:true, user:a, bootstrap:bootstrap_(a) };
}

function resolveTeam_(emp, configs) {
  const none = { team:'Unassigned', zoneLead:'', process:'' };
  const d = key_(emp.department), o = key_(emp.office), m = key_(emp.manager);
  let best = null, bestScore = -1, tie = false;
  configs.forEach(function(c) {
    let s = 0;
    if (key_(c.department)) { if (key_(c.department) !== d) return; s += 4; }
    if (key_(c.office))     { if (key_(c.office) !== o) return; s += 3; }
    if (key_(c.manager))    { if (key_(c.manager) !== m) return; s += 2; }
    if (s > bestScore) { best = c; bestScore = s; tie = false; }
    else if (s === bestScore && best && key_(best.team) !== key_(c.team)) tie = true;
  });
  if (!best || tie) return none;
  return { team: best.team || 'Unassigned', zoneLead: best.zoneLead, process: best.process };
}

const MONTHS_ = ['January','February','March','April','May','June','July','August','September','October','November','December'];
// Period filter: Year + Month. Empty = live (today).
function period_(f) {
  f = f || {};
  const y = Number(f.year || 0), m = Number(f.month || 0), now = new Date();
  if (!y && !m) return { live:true, year:now.getFullYear(), label:'This month (Live)' };
  const yr = y || now.getFullYear();
  return { live:false, year:yr,
    start: m ? new Date(yr, m - 1, 1) : new Date(yr, 0, 1),
    end:   m ? new Date(yr, m, 1)     : new Date(yr + 1, 0, 1),
    label: m ? MONTHS_[m - 1] + ' ' + yr : 'Year ' + yr };
}
// Status "as of" the end of the selected period
function viewEmps_(emps, per) {
  if (per.live) return emps;
  return emps.filter(function(e){ return !e.doj || e.doj < per.end; }).map(function(e){
    const c = Object.assign({}, e);
    c.status = (e.deactiveDate ? e.deactiveDate >= per.end : isActive_(e)) ? 'Active' : 'Inactive';
    if (e.salaryHist) c.salary = salaryAsOf_(e.salaryHist, new Date(per.end.getTime() - 1), e.salary);
    return c;
  });
}
function desigBucket_(d) {
  d = key_(d);
  if (d.indexOf('zone lead') > -1) return 'zoneLead';
  if (/senior associate|sr\.? associate/.test(d)) return 'seniorAssociate';
  if (d.indexOf('associate') > -1) return 'associate';
  if (d.indexOf('operator') > -1 || d.indexOf('sorter') > -1) return 'operator';
  if (d.indexOf('executive') > -1) return 'executive';
  return '';
}

function attendanceDayOffSummary_(per, f) {
  try {
    const emps=(f&&f.employees)||[], ids={};
    emps.forEach(function(e){ ids[key_(e.employeeId)]=1; });
    const months=[];
    if(per && per.live) months.push(cycleKey_(new Date()));
    else if(per && per.start && per.end){
      // Month-selected dashboard: use the matching payroll attendance cycle.
      if(per.start.getMonth()===per.end.getMonth()-1 || per.start.getFullYear()!==per.end.getFullYear()) {
        const d=new Date(per.start.getFullYear(),per.start.getMonth()+1,1); months.push(cycleKey_(new Date(d.getFullYear(),d.getMonth(),20)));
      } else {
        for(let m=0;m<12;m++) months.push(String(per.year)+'-'+pad2_(m+1));
      }
    }
    let total=0;
    months.forEach(function(mk){ ['Operator','Management'].forEach(function(g){
      const t=readAtt_(mk,g); if(!t)return;
      t.rows.forEach(function(r){ if(!ids[key_(r.id)])return; const z=attendanceRowCounts_(r,t); total+=z.offDay; });
    }); });
    return total;
  } catch(e) { return 0; }
}
function makeDashboard_(emps, cfgs, a, per, all, filters) {
  per = per || period_({}); all = all || emps;
  const active = emps.filter(isActive_), inactive = emps.filter(function(e){ return !isActive_(e); });
  const now = new Date(), sal = active.map(function(e){return e.salary;}).filter(function(v){return v > 0;});
  const payroll = sal.reduce(function(x,y){return x+y;}, 0);
  const inP = function(d){ if (!d) return false; return per.live ? sameMonth_(d, now) : (d >= per.start && d < per.end); };
  const joined = all.filter(function(e){ return inP(e.doj); }).length;
  const left = all.filter(function(e){ return inP(e.deactiveDate); }).length;
  const des = { executive:0, seniorAssociate:0, associate:0, operator:0, zoneLead:0 };
  active.forEach(function(e){ const b = desigBucket_(e.designation); if (b) des[b]++; });
  const off = {};
  active.forEach(function(e){ const k = e.office || 'Unassigned'; off[k] = (off[k] || 0) + 1; });
  let expired = 0, soon = 0, missing = 0;
  active.forEach(function(e){
    if (e.contractDate) {
      const d = Math.ceil((e.contractDate.getTime() - now.getTime()) / 86400000);
      if (d < 0) expired++; else if (d <= 30) soon++;
    }
    if (!e.department || !e.manager || !e.doj || !e.salary) missing++;
  });
  return {
    period: { label: per.label, live: per.live, year: per.year },
    kpis: { total:emps.length, active:active.length, deactive:inactive.length, dayOff:attendanceDayOffSummary_(per, filtersForAttendance_(all, filters)),
      joinThisMonth:joined, deactiveThisMonth:left,
      attritionPct: active.length ? left / active.length * 100 : 0,
      payroll: a.salaryView ? payroll : null,
      averageSalary: a.salaryView ? (sal.length ? payroll / sal.length : 0) : null },
    alerts: { contractExpired:expired, contractSoon:soon, missingData:missing },
    byOffice: Object.keys(off).map(function(k){ return { office:k, count:off[k] }; }).sort(function(x,y){ return y.count - x.count; }),
    designation: des,
    workforce: workforce_(emps),
    capacity: capacity_(emps, cfgs),
    joiningTrend: trend_(all, 'doj', per.year),
    deactivationTrend: trend_(all, 'deactiveDate', per.year)
  };
}
function workforce_(emps) {
  const map = {};
  emps.forEach(function(e) {
    const k = e.department || 'Unassigned';
    const x = map[k] || (map[k] = { department:k, total:0, executive:0, seniorAssociate:0, associate:0, operator:0, active:0, deactive:0, dayOff:0 });
    x.total++; isActive_(e) ? x.active++ : x.deactive++;
    const b = desigBucket_(e.designation); if (b && b !== 'zoneLead') x[b]++;
  });
  return Object.keys(map).sort().map(function(k){ return map[k]; });
}
function cfgKey_(c) { return [key_(c.department), key_(c.team), key_(c.office), key_(c.manager)].join('|'); }
function capacity_(emps, cfgs) {
  const groups = {};
  cfgs.forEach(function(c) {
    const k = cfgKey_(c);
    const g = groups[k] || (groups[k] = { department:c.department||'Unassigned', team:c.team||'Unassigned', office:c.office,
      zoneLead:c.zoneLead, manager:c.manager, process:c.process, op:0, ptnRate:0, bulkRate:0, bulkPct:0,
      shiftHours:0, breakHours:0, ectHours:0, links:[c.link1,c.link2,c.link3,c.link4].filter(Boolean) });
    ['ptnRate','bulkRate','bulkPct','shiftHours','breakHours','ectHours'].forEach(function(f){ g[f] = c[f] || g[f]; });
  });
  emps.forEach(function(e) {
    if (!isActive_(e) || !isOperator_(e)) return;
    const c = findCfg_(e, cfgs);
    if (c && groups[cfgKey_(c)]) groups[cfgKey_(c)].op++;
  });
  return Object.keys(groups).map(function(k) {
    const g = groups[k], avail = g.op, ptnPct = Math.max(0, 1 - g.bulkPct);
    const ptn = avail * g.ptnRate * ptnPct, bulk = avail * g.bulkRate * g.bulkPct, hourly = ptn + bulk;
    const normal = hourly * Math.max(0, g.shiftHours - g.breakHours), ect = hourly * g.ectHours;
    return { department:g.department, team:g.team, office:g.office, zoneLead:g.zoneLead, manager:g.manager, process:g.process,
      sorterOperator:g.op, dayOff:0, available:avail, ptnRate:g.ptnRate, bulkRate:g.bulkRate, bulkPct:g.bulkPct,
      currentPTN:ptn, currentBulk:bulk, processHourly:hourly, shiftHours:g.shiftHours, breakHours:g.breakHours,
      ectHours:g.ectHours, normalCapacity:normal, ectCapacity:ect, maximumCapacity:normal + ect,
      currentOrder:null, configComplete:g.ptnRate > 0 && g.bulkRate > 0 && g.shiftHours > 0, links:g.links };
  }).sort(function(x, y){ return String(x.team).localeCompare(String(y.team)); });
}
function findCfg_(emp, cfgs) {
  const m = resolveTeam_(emp, cfgs);
  if (m.team === 'Unassigned') return null;
  const c = cfgs.filter(function(x){ return key_(x.team) === key_(m.team); });
  if (!c.length) return null;
  return c.filter(function(x){
    return key_(x.department) === key_(emp.department) && key_(x.office) === key_(emp.office) && key_(x.manager) === key_(emp.manager);
  })[0] || c[0];
}
function applyFilters_(emps, f) {
  const chk = function(want, have){ want = key_(want); return !want || want === 'all' || want === key_(have); };
  return emps.filter(function(e){
    return chk(f.department, e.department) && chk(f.team, e.team) && chk(f.office, e.office) && chk(f.status, e.status);
  });
}
function trend_(emps, field, year) {
  const y = year || new Date().getFullYear(), out = [];
  for (let m = 0; m < 12; m++) {
    const s = new Date(y, m, 1), n = new Date(y, m + 1, 1);
    out.push({ month: Utilities.formatDate(s, tz_(), 'MMM'),
      count: emps.filter(function(e){ return e[field] && e[field] >= s && e[field] < n; }).length });
  }
  return out;
}

/* ================= TURNOVER ================= */
function getTurnover(token, f) {
  const a = require_(token);
  f = f || {};
  const year = Number(f.year || new Date().getFullYear()), month = Number(f.month || 0);
  const base = applyFilters_(readEmployees_(), { department:f.department });
  const emps = applyFilters_(base, { team:f.team });
  const calc = function(list, s, n) {
    const opening = list.filter(function(e){ return e.doj && e.doj < s && !(e.deactiveDate && e.deactiveDate < s); }).length;
    const join = list.filter(function(e){ return e.doj && e.doj >= s && e.doj < n; }).length;
    const resigned = list.filter(function(e){ return e.deactiveDate && e.deactiveDate >= s && e.deactiveDate < n; }).length;
    const closing = opening + join - resigned, avg = (opening + closing) / 2;
    return { opening:opening, join:join, resigned:resigned, active:closing, turnover: avg > 0 ? resigned / avg * 100 : 0 };
  };
  const rows = [];
  for (let m = 0; m < 12; m++) {
    const s = new Date(year, m, 1), n = new Date(year, m + 1, 1);
    rows.push(Object.assign({ idx:m + 1, month:Utilities.formatDate(s, tz_(), 'MMM') }, calc(emps, s, n)));
  }
  const ps = month ? new Date(year, month - 1, 1) : new Date(year, 0, 1);
  const pn = month ? new Date(year, month, 1) : new Date(year + 1, 0, 1);
  const byTeam = {};
  // Team-wise breakdown must use Data sheet column L (Department), exposed as hrDepartment.
  base.forEach(function(e){ const k = e.hrDepartment || 'Unassigned'; (byTeam[k] = byTeam[k] || []).push(e); });
  const teams = Object.keys(byTeam).sort().map(function(k){ return Object.assign({ team:k }, calc(byTeam[k], ps, pn)); })
    .filter(function(t){ return t.opening || t.join || t.resigned; });
  log_(a.email, 'TURNOVER_VIEW', JSON.stringify(f));
  return { year:year, month:month, periodLabel: month ? MONTHS_[month - 1] + ' ' + year : 'Year ' + year, rows:rows, teams:teams, teamBreakdownSource:'Data sheet L:L (Department)' };
}

function calculateForecast(token, p) {
  const a = require_(token);
  p = p || {};
  const single = p.method === 'single';
  const orders = toNum_(p.totalOrders), hours = toNum_(p.workingHours);
  let ptnPct = pct_(p.ptnPct), bulkPct = pct_(p.bulkPct), ptnRate = toNum_(p.ptnRate), bulkRate = toNum_(p.bulkRate);
  const ops = toNum_(p.activeOperators), off = toNum_(p.dayOff), ect = toNum_(p.ectHours), buf = toNum_(p.safetyBuffer);
  if (single) { ptnPct = 1; bulkPct = 0; bulkRate = ptnRate; }
  if (orders <= 0) throw new Error('Total Orders must be greater than 0.');
  if (hours <= 0) throw new Error('Working Hours must be greater than 0.');
  if (single && ptnRate <= 0) throw new Error('Orders / Resource / Hour must be greater than 0.');
  if (!single && (ptnRate <= 0 || bulkRate <= 0)) throw new Error('PTN and BULK capacity/operator/hour must be greater than 0.');
  if (!single && Math.abs(ptnPct + bulkPct - 1) > 0.01) throw new Error('PTN % + BULK % must be equal to 100.');
  const hourly = orders / hours, ptnH = hourly * ptnPct, bulkH = hourly * bulkPct;
  const reqPtn = Math.ceil(ptnH / ptnRate), reqBulk = Math.ceil(bulkH / bulkRate), req = reqPtn + reqBulk;
  const avail = Math.max(0, ops - off), perHour = avail * (ptnRate * ptnPct + bulkRate * bulkPct);
  const normal = perHour * hours, ectCap = perHour * ect, max = normal + ectCap;
  let status = 'Normal Capacity Sufficient';
  if (normal < orders) status = max >= orders ? 'ECT Required' : 'Manpower / Capacity Shortage';
  log_(a.email, 'FORECAST_RUN', JSON.stringify(p));
  return { method:single ? 'single' : 'ptnbulk', totalOrders:orders, hourlyRequirement:hourly, ptnHourly:ptnH, bulkHourly:bulkH,
    requiredPTNOperators:reqPtn, requiredBulkOperators:reqBulk, requiredManpower:req, activeOperators:ops, dayOff:off,
    available:avail, manpowerGap:req - avail, additionalResource:Math.max(0, req + buf - avail), normalCapacity:normal,
    ectCapacity:ectCap, maximumCapacity:max, utilization: normal > 0 ? orders / normal * 100 : 0, status:status };
}

/* ================= REPORTS ================= */
function getReports(token, filters) {
  const a = require_(token);
  const emps = applyFilters_(readEmployees_(), filters || {}), active = emps.filter(isActive_), now = new Date();
  // salary
  let salary = { allowed:false };
  if (a.salaryView) {
    const sal = active.map(function(e){return e.salary;}).filter(function(v){return v > 0;}), byD = {};
    active.forEach(function(e){
      const k = e.department || 'Unassigned', x = byD[k] || (byD[k] = { department:k, headcount:0, payroll:0, average:0 });
      x.headcount++; x.payroll += e.salary || 0;
    });
    salary = { allowed:true, activePayroll:sal.reduce(function(x,y){return x+y;},0), average:average_(sal),
      min:sal.length ? Math.min.apply(null, sal) : 0, max:sal.length ? Math.max.apply(null, sal) : 0,
      byDepartment:Object.keys(byD).map(function(k){ const x = byD[k]; x.average = x.headcount ? x.payroll / x.headcount : 0; return x; })
        .sort(function(x, y){ return y.payroll - x.payroll; }) };
  }
  // aging
  const aging = { '<3m':0, '3-6m':0, '6-12m':0, '1-2y':0, '2y+':0 };
  active.forEach(function(e){
    if (!e.doj) return;
    const m = monthDiff_(e.doj, now);
    aging[m < 3 ? '<3m' : m < 6 ? '3-6m' : m < 12 ? '6-12m' : m < 24 ? '1-2y' : '2y+']++;
  });
  // contract
  const contract = { expired:0, within7:0, within30:0, within60:0, within90:0 };
  active.forEach(function(e){
    if (!e.contractDate) return;
    const d = Math.ceil((e.contractDate.getTime() - now.getTime()) / 86400000);
    if (d < 0) contract.expired++; else if (d <= 7) contract.within7++; else if (d <= 30) contract.within30++;
    else if (d <= 60) contract.within60++; else if (d <= 90) contract.within90++;
  });
  // office / manager
  const off = {}, mgr = {};
  emps.forEach(function(e){
    const k = e.office || 'Unassigned', x = off[k] || (off[k] = { office:k, total:0, active:0, deactive:0 });
    x.total++; isActive_(e) ? x.active++ : x.deactive++;
  });
  active.forEach(function(e){ const k = e.manager || 'Unassigned'; mgr[k] = (mgr[k] || 0) + 1; });
  // data quality
  const ids = {};
  emps.forEach(function(e){ const k = key_(e.employeeId); if (k) ids[k] = (ids[k] || 0) + 1; });
  let dup = 0; Object.keys(ids).forEach(function(k){ if (ids[k] > 1) dup += ids[k]; });
  const valid = emps.filter(function(e){
    return [e.employeeId, e.name, e.designation, e.department, e.doj, e.salary, e.manager].every(function(v){ return v && v !== 0; });
  }).length;
  const filled = function(f){ return active.filter(function(e){ return normText_(e[f]) !== ''; }).length; };
  log_(a.email, 'REPORTS_VIEW', JSON.stringify(filters || {}));
  return { salary:salary, aging:aging, contract:contract,
    office:Object.keys(off).map(function(k){return off[k];}).sort(function(x,y){return y.total - x.total;}),
    manager:Object.keys(mgr).map(function(k){ return { manager:k, active:mgr[k] }; }).sort(function(x,y){return y.active - x.active;}),
    dataQuality:{ total:emps.length, valid:valid, incomplete:emps.length - valid, duplicateEmployeeId:dup,
      health: emps.length ? valid / emps.length * 100 : 100 },
    assets:{ laptop:filled('laptop'), serial:filled('serial'), mouse:filled('mouse'), keyboard:filled('keyboard'),
      headphone:filled('headphone'), idCard:filled('idCard'), vest:filled('vest') } };
}

function monthKeyFromLabel_(v){
  const s = normText_(v).replace(/\s+/g,' ');
  if (!s) return '';
  let m = /^(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)[a-z]*[- ](\d{2,4})$/i.exec(s);
  if (!m) return '';
  const mi = MON3_.indexOf(m[1].slice(0,3).toLowerCase());
  let y = Number(m[2]); if (y < 100) y += 2000;
  return mi >= 0 ? y + '-' + pad2_(mi + 1) : '';
}
function readMomOrder_(){

  // Flexible MOM Order parser. It supports both:
  // 1) Row with "Department" + month headers on the same row, and
  // 2) Month headers on a separate row, with Department on another row.
  // It also ignores helper rows such as "Order Qty".
  const v = sh.getDataRange().getDisplayValues();
  const maxScan = Math.min(12, v.length);
  const deptAliases = ['department','dept'];
  let deptRow = -1, monthHeaderRow = -1, bestMonthCount = 0;

  for (let r = 0; r < maxScan; r++) {
    const first = key_(v[r][0]);
    if (deptRow < 0 && deptAliases.indexOf(first) > -1) deptRow = r;
    let cnt = 0;
    for (let c = 1; c < v[r].length; c++) if (monthKeyFromLabel_(v[r][c])) cnt++;
    if (cnt > bestMonthCount) { bestMonthCount = cnt; monthHeaderRow = r; }
  }
  if (deptRow < 0) throw new Error('Department header not found in the MOM Order sheet.');
  if (monthHeaderRow < 0 || bestMonthCount === 0) throw new Error('Month headers (e.g. May-26, Jun-26) not found in the MOM Order sheet.');

  const months = [];
  for (let c = 1; c < v[monthHeaderRow].length; c++) {
    const mk = monthKeyFromLabel_(v[monthHeaderRow][c]);
    if (!mk) continue;
    months.push({ col:c, key:mk, label:normText_(v[monthHeaderRow][c]), type:'order qty' });
  }

  // Data starts after both the Department header and month header rows.
  // Rows whose first cell is Order Qty / Total / blank are ignored.
  const dataStart = Math.max(deptRow, monthHeaderRow) + 1;
  const rows = [], map = {}, totalMap = {};

  for (let r = dataStart; r < v.length; r++) {
    const dept = normText_(v[r][0]);
    const dk = key_(dept);
    if (!dept) continue;
    const hasNumber = months.some(function(m){ return normText_(v[r][m.col]) !== ''; });
    if (!hasNumber) continue;

    // If the source sheet has an explicit Total Order row in column A,
    // keep that value separately. This value is used ONLY for the Total Order card.
    const totalLabel = key_(dept).replace(/\s+/g,' ');
    if (/^(total order|total orders|grand total order|grand total)$/.test(totalLabel)) {
      months.forEach(function(m){ totalMap[m.key] = toNum_(v[r][m.col]); });
      continue;
    }
    if (/^(order\s*qty|total|department|dept)$/.test(totalLabel)) continue;

    const row = { department:dept, values:{} };
    months.forEach(function(m){
      row.values[m.key] = toNum_(v[r][m.col]);
      map[m.key+'|'+dk] = row.values[m.key] || 0;
    });
    rows.push(row);
  }
  const uniqMonths = months.filter(function(m,i,a){ return a.findIndex(function(x){return x.key===m.key;})===i; }).sort(function(a,b){return a.key.localeCompare(b.key);});
  return (_mem.momOrder = { months:uniqMonths, rows:rows, map:map, totalMap:totalMap, sheet:sh.getName(), deptRow:deptRow+1, monthHeaderRow:monthHeaderRow+1 });
}

function attendanceDepartmentStats_(monthKey, department){
  const parts=String(monthKey || '').split('-'), year=Number(parts[0]), month=Number(parts[1]);
  if (!year || !month) return {resourceCount:0,presentDays:0,dayOff:0,weekend:0,leave:0,absent:0,holiday:0};
  const empMap={}; readEmployees_().forEach(function(e){ empMap[key_(e.employeeId)]=e; });
  const want=key_(department), seen={}, out={resourceCount:0,presentDays:0,dayOff:0,weekend:0,leave:0,absent:0,holiday:0};
  ['Operator','Management'].forEach(function(g){
    const t=readAtt_(monthKey,g); if(!t) return;
    t.rows.forEach(function(r){

      if(want && want!=='all' && key_(dept)!==want && key_(deptL)!==want) return;
      const id=key_(r.id); if(!id || seen[id]) return; seen[id]=1; out.resourceCount++;
      for(let i=0;i<t.dates.length;i++){
        const raw=String(r.codes[i]||'').trim().toLowerCase(), cl=String(r.classes[i]||'').toUpperCase();

      }
    });
  });

}

const MON3_ = ['jan','feb','mar','apr','may','jun','jul','aug','sep','oct','nov','dec'];
function pad2_(n) { return (n < 10 ? '0' : '') + n; }
// Payroll cycle: 21st -> next month 20th. Tab "Oct 26" = 21 Sep 2026 .. 20 Oct 2026
function cycleKey_(d) {
  let y = d.getFullYear(), m = d.getMonth();
  if (d.getDate() >= 21) { m++; if (m > 11) { m = 0; y++; } }
  return y + '-' + pad2_(m + 1);
}
function attTabs_() {
  if (_mem.attTabs) return _mem.attTabs;
  let ss;
  try { ss = SpreadsheetApp.openById(APP.ATT_ID); }
  catch (e) { throw new Error('The Attendance spreadsheet could not be opened. Please verify that the script account has access.\n' + e.message); }
  _mem.attTabs = ss.getSheets().map(function(s) {
    const m = /^([A-Za-z]{3})[A-Za-z]*[\s'-]*(\d{2,4})\s*\(\s*(operator|management)\s*\)/i.exec(s.getName().trim());
    if (!m) return null;
    const mon = MON3_.indexOf(m[1].toLowerCase());
    if (mon < 0) return null;
    let yr = Number(m[2]); if (yr < 100) yr += 2000;
    return { sheet:s, name:s.getName(), month:mon, year:yr, key:yr + '-' + pad2_(mon + 1),
      group:m[3].charAt(0).toUpperCase() + m[3].slice(1).toLowerCase() };
  }).filter(Boolean);
  return _mem.attTabs;
}
function classify_(raw) {
  const c = String(raw || '').trim().toLowerCase();
  if (!c) return '';
  if (/^(p|present)\b/.test(c) || /^\d/.test(c)) return 'P';
  if (/^(a|absent)\b/.test(c)) return 'A';
  if (/^(l|leave|cl|sl|al|el|ml|lwp)\b/.test(c)) return 'L';
  if (/^(off|o|wo|w|weekend|weekly off)\b/.test(c)) return 'OFF';
  if (/^(ph|h|holiday|public)/.test(c)) return 'H';
  return 'X';
}
function sumKey_(h) {
  h = String(h).replace(/\s+/g, ' ').trim().toLowerCase();
  const map = { 'present':'present', 'absent':'absent', 'leave':'leave', 'off day':'offDay', 'holiday':'holiday', 'ot':'ot',
    'late':'late', 'early':'early', 'night shift':'nightShift', 'l/e time':'leTime', 'cut days':'cutDays', 'off day duty':'offDayDuty', 'total':'total' };
  if (map[h]) return map[h];
  if (/pub.*holiday/.test(h)) return 'publicHoliday';
  if (/remark/.test(h)) return 'remarks';
  return '';
}
function parseAtt_(tab) {
  const v = tab.sheet.getDataRange().getDisplayValues();
  let hr = -1;
  for (let i = 0; i < Math.min(v.length, 15); i++) {
    if (key_(v[i][0]) === 'id' && v[i].some(function(c){ return /^name$/i.test(String(c).trim()); })) { hr = i; break; }
  }
  if (hr < 0) throw new Error('Attendance header row (ID, Name, ...) not found: ' + tab.name);
  const head = v[hr].map(function(x){ return String(x).replace(/\s+/g, ' ').trim(); });
  const col = function(re){ for (let i = 0; i < head.length; i++) if (re.test(head[i])) return i; return -1; };
  const c = { id:Math.max(0, col(/^id$/i)), name:col(/^name$/i), des:col(/^designation$/i), dept:col(/^department$/i),
    doj:col(/^doj$/i), res:col(/^resignation/i), wk:col(/^week/i) };
  const dates = [], dre = /^(\d{1,2})[-\/ ]*([A-Za-z]{3})/;
  let j = c.wk >= 0 ? c.wk + 1 : 0;
  for (; j < head.length; j++) {
    const m = dre.exec(head[j]);
    if (!m) { if (dates.length) break; continue; }
    const mon = MON3_.indexOf(m[2].toLowerCase());
    if (mon < 0) { if (dates.length) break; continue; }
    const yr = mon === tab.month ? tab.year : (mon > tab.month ? tab.year - 1 : tab.year);
    const dt = new Date(yr, mon, Number(m[1]));
    dates.push({ col:j, iso:iso_(dt), label:Number(m[1]) + ' ' + MONTHS_[mon].slice(0, 3), dow:dt.getDay() });
  }
  const sumCols = [];
  for (let k = j; k < head.length; k++) { const key = sumKey_(head[k]); if (key) sumCols.push({ col:k, key:key }); }
  const get = function(row, i){ return i >= 0 ? String(row[i] || '').trim() : ''; };
  const rows = [];
  for (let r = hr + 1; r < v.length; r++) {
    const row = v[r], id = get(row, c.id), name = get(row, c.name);
    if (!id && !name) continue;
    const codes = dates.map(function(d){ return String(row[d.col] || '').trim(); });
    const classes = codes.map(classify_);
    const counts = { P:0, A:0, L:0, OFF:0, H:0 };
    classes.forEach(function(x){ if (counts[x] !== undefined) counts[x]++; });
    const summary = {};
    sumCols.forEach(function(s){
      const raw = String(row[s.col] || '').trim();
      summary[s.key] = (s.key === 'remarks' || s.key === 'leTime') ? raw : (raw === '' ? null : toNum_(raw));
    });
    rows.push({ id:id, name:name, designation:get(row, c.des), department:get(row, c.dept), doj:get(row, c.doj),
      resign:get(row, c.res), weekend:get(row, c.wk), codes:codes, classes:classes, counts:counts, summary:summary });
  }
  return { tab:tab.name, key:tab.key, group:tab.group,
    dates:dates.map(function(d){ return { iso:d.iso, label:d.label, dow:d.dow }; }), rows:rows };
}
function readAtt_(key, group) {
  const mk = 'att|' + key + '|' + group;
  if (_mem[mk] !== undefined) return _mem[mk];
  const tab = attTabs_().filter(function(t){ return t.key === key && t.group === group; })[0];
  return (_mem[mk] = tab ? parseAtt_(tab) : null);
}

    if (!t) continue;
    const r = t.rows.filter(function(x){ return key_(x.id) === id; })[0];
    if (r) return { found:true, key:key, group:groups[i], tab:t.tab, dates:t.dates, codes:r.codes, classes:r.classes,
      counts:r.counts, summary:r.summary };
  }
  return { found:false, key:key, message:'No attendance found for Employee ID ' + employeeId + ' in this cycle (' + key + ').' };
}
function weekendParts_(v) {
  const s = key_(v);
  if (!s) return [];
  const names = {sunday:0,monday:1,tuesday:2,wednesday:3,thursday:4,friday:5,saturday:6,
    sun:0,mon:1,tue:2,tues:2,wed:3,weds:3,thu:4,thur:4,thurs:4,fri:5,sat:6};
  const out = [];
  s.replace(/[&,/|+]+/g,' ').replace(/\s*-\s*/g,' ').split(/\s+/).forEach(function(x){
    if (names[x] !== undefined && out.indexOf(names[x]) < 0) out.push(names[x]);
    else if (/^[0-6]$/.test(x)) { const n=Number(x); if(out.indexOf(n)<0)out.push(n); }
  });
  return out;
}
function isWeekend_(weekend, dow) { return weekendParts_(weekend).indexOf(Number(dow)) > -1; }
function attendanceRowCounts_(r,t) {
  const out = {present:0,absent:0,leave:0,offDay:0,weekend:0,holiday:0,late:0,ot:0,early:0,nightShift:0,cutDays:0};
  const s=r.summary||{};
  out.late=Number(s.late)||0; out.ot=Number(s.ot)||0; out.early=Number(s.early)||0;
  out.nightShift=Number(s.nightShift)||0; out.cutDays=Number(s.cutDays)||0;

  for(let i=0;i<t.dates.length;i++){
    const raw=String(r.codes[i]||'').trim().toLowerCase();
    const cl=String(r.classes[i]||'').toUpperCase();
    const wk=isWeekend_(r.weekend,t.dates[i].dow);
    const code = cl || (raw ? classify_(raw) : '');

    // Priority: actual attendance status first. A/L/H on a weekend must NOT
    // become Day Off. Weekend Worked means the employee actually worked (P).
    if(code==='P'){
      out.present++;
      if(wk) out.weekend++;
      continue;
    }
    if(code==='A'){ out.absent++; continue; }
    if(code==='L'){ out.leave++; continue; }
    if(code==='H'){ out.holiday++; continue; }

    // OFF / blank roster day on the employee's Weekend/Week = Day Off.
    if(code==='OFF' || wk){ out.offDay++; continue; }
  }

  // Keep sheet summary for Present/Absent/Leave/Holiday only when available.
  // Day Off and Weekend Worked remain calculated from daily roster/code data.
  if(s.present!=null) out.present=Number(s.present)||out.present;
  if(s.absent!=null) out.absent=Number(s.absent)||out.absent;
  if(s.leave!=null) out.leave=Number(s.leave)||out.leave;
  if(s.holiday!=null || s.publicHoliday!=null) out.holiday=(Number(s.holiday)||0)+(Number(s.publicHoliday)||0);
  return out;
}

  const key=monthKey||cycleKey_(new Date()), groups=group&&group!=='All'?[group]:['Operator','Management'], set={};
  groups.forEach(function(g){ const t=readAtt_(key,g); if(!t)return; t.rows.forEach(function(r){ if(r.weekend) set[normText_(r.weekend)]=1; }); });

}
function getAttendanceReport(token, f) {
  require_(token);
  f = f || {};
  const key = f.key || cycleKey_(new Date()), group = f.group || 'All';
  const groups = group === 'All' ? ['Operator', 'Management'] : [group];
  const empMap = {}; readEmployees_().forEach(function(e){ empMap[key_(e.employeeId)] = e; });
  const q = key_(f.search), want = key_(f.department), wantWeekend=key_(f.weekend);
  const rows = [], missing = [];
  groups.forEach(function(g) {
    const t = readAtt_(key, g);
    if (!t) { missing.push(g); return; }
    t.rows.forEach(function(r) {
      const e = empMap[key_(r.id)], dept = (e ? e.department : r.department) || 'Unassigned';
      if (want && want !== 'all' && key_(dept) !== want) return;
      if (wantWeekend && wantWeekend !== 'all' && key_(r.weekend) !== wantWeekend) return;
      if (q && [r.id, r.name, dept, r.weekend].join(' ').toLowerCase().indexOf(q) < 0) return;
      const z=attendanceRowCounts_(r,t);
      rows.push({ id:r.id, name:r.name, designation:r.designation, department:dept, group:g, weekend:r.weekend,
        present:z.present, absent:z.absent, leave:z.leave, offDay:z.offDay, weekendDays:z.weekend, holiday:z.holiday,
        ot:z.ot, late:z.late, early:z.early, nightShift:z.nightShift, cutDays:z.cutDays,
        total:r.summary.total == null ? '' : r.summary.total });
    });
  });
  const add = function(x, r){ ['present','absent','leave','offDay','weekendDays','late','ot'].forEach(function(k){ x[k] += Number(r[k]) || 0; }); };
  const mk = function(n){ return { department:n, headcount:0, present:0, absent:0, leave:0, offDay:0, weekendDays:0, late:0, ot:0, absentPct:0 }; };
  const totals = mk('All'), depts = {};
  rows.forEach(function(r){ totals.headcount++; add(totals, r); const d = depts[r.department] || (depts[r.department] = mk(r.department)); d.headcount++; add(d, r); });
  const pct = function(x){ const b=x.present+x.absent+x.leave; x.absentPct=b?x.absent/b*100:0; };
  pct(totals);
  Object.keys(depts).forEach(function(k){pct(depts[k]);});
  const parts = key.split('-'), yr = Number(parts[0]), mo = Number(parts[1]);
  return { key:key, label:MONTHS_[mo - 1] + ' ' + yr, missing:missing, totals:totals, rows:rows,
    departments:Object.keys(depts).sort().map(function(k){ return depts[k]; }) };
}
// Today's off-day + leave count for a set of employee IDs (used by Forecast)
function attDayOff_(ids) {
  try {
    const today = new Date(), iso = iso_(today), key = cycleKey_(today);
    let off = 0, found = false;
    ['Operator', 'Management'].forEach(function(g) {
      const t = readAtt_(key, g);
      if (!t) return;
      let idx = -1;
      t.dates.forEach(function(d, i){ if (d.iso === iso) idx = i; });
      if (idx < 0) return;
      found = true;
      t.rows.forEach(function(r){ if (ids[key_(r.id)]) { const z=attendanceRowCounts_(r,t); if(z.offDay>0 || r.classes[idx] === 'L') off++; } });
    });
    return found ? off : null;
  } catch (e) { return null; }
}

// Reads the 'Salary History' sheet (LOG spreadsheet). Backend-e manually row dile-o salary update hobe.
// Required columns: B=Employee ID, E=New Salary, F=Effective Date.  (D=Old Salary optional)
function readSalaryHistoryMap_() {
  if (_mem.salHist) return _mem.salHist;
  const map = {};
  try {
    const sh = audit_().getSheetByName(APP.SALARY_SHEET);
    if (sh && sh.getLastRow() > 1) {
      sh.getRange(2, 1, sh.getLastRow() - 1, 8).getValues().forEach(function(r, i) {
        const id = key_(r[1]), sal = toNum_(r[4]);
        if (!id || !(sal > 0)) return;
        const eff = toDate_(r[5]) || toDate_(r[0]);
        if (!eff) return;
        (map[id] = map[id] || []).push({ eff:eff, sal:sal, old:toNum_(r[3]), ts:toDate_(r[0]), seq:i });
      });
      Object.keys(map).forEach(function(k){
        map[k].sort(function(x, y){ return (x.eff - y.eff) || ((x.ts ? x.ts.getTime() : 0) - (y.ts ? y.ts.getTime() : 0)) || (x.seq - y.seq); });
      });
    }
  } catch (e) { console.error('Salary history read failed: ' + e.message); }
  return (_mem.salHist = map);
}
function salaryAsOf_(hist, date, fallback) {
  let cur = null;
  hist.forEach(function(h){ if (h.eff <= date) cur = h; });
  if (cur) return cur.sal;
  return hist.length && hist[0] && hist[0].old > 0 ? hist[0].old : fallback;   // before the first recorded change
}

/* ================= UTILS ================= */
function normText_(v) { return v === null || v === undefined ? '' : String(v).trim(); }
function normEmail_(v) { return normText_(v).toLowerCase(); }
function key_(v) { return normText_(v).toLowerCase().replace(/\s+/g, ' '); }
function toBool_(v) { return v === true || v === 1 || ['true','yes','y','1','active','enabled'].indexOf(key_(v)) > -1; }
function toNum_(v) {
  if (v === null || v === undefined || v === '') return 0;
  if (typeof v === 'number') return isFinite(v) ? v : 0;
  const n = Number(String(v).replace(/,/g, '').replace(/[^\d.-]/g, ''));
  return isFinite(n) ? n : 0;
}
function pct_(v) {
  if (v === null || v === undefined || v === '') return 0;
  const n = typeof v === 'number' ? v : Number(String(v).replace('%', '').trim());
  if (!isFinite(n)) return 0;
  return n > 1 ? n / 100 : n;
}
function url_(v) { const s = normText_(v); return /^https?:\/\/.+/i.test(s) ? s : ''; }
function toDate_(v) {
  if (!v) return null;
  const d = Object.prototype.toString.call(v) === '[object Date]' ? new Date(v) : new Date(v);
  return isNaN(d.getTime()) ? null : d;
}
function plain_(v) { return Object.prototype.toString.call(v) === '[object Date]' ? iso_(v) : (v === null || v === undefined ? '' : v); }
function tz_() { return Session.getScriptTimeZone() || 'Asia/Dhaka'; }
function iso_(d) { return d ? Utilities.formatDate(d, tz_(), 'yyyy-MM-dd') : ''; }
function dt_(d) { return d && d.getTime ? Utilities.formatDate(d, tz_(), 'yyyy-MM-dd HH:mm:ss') : String(d || ''); }
function isActive_(e) { return key_(e.status) === 'active'; }
function isOperator_(e) { const d = key_(e.designation); return d.indexOf('operator') > -1 || d.indexOf('sorter') > -1; }
function sameMonth_(a, b) { return !!a && !!b && a.getFullYear() === b.getFullYear() && a.getMonth() === b.getMonth(); }
function monthDiff_(s, e) { return (e.getFullYear() - s.getFullYear()) * 12 + e.getMonth() - s.getMonth(); }
function average_(arr) { return arr && arr.length ? arr.reduce(function(x, y){ return x + y; }, 0) / arr.length : 0; }
function uniq_(arr) { return Array.from(new Set(arr.filter(Boolean))).sort(); }
function yearList_(emps) {
  const y = {};
  emps.forEach(function(e){ if (e.doj) y[e.doj.getFullYear()] = 1; if (e.deactiveDate) y[e.deactiveDate.getFullYear()] = 1; });
  y[new Date().getFullYear()] = 1;
  return Object.keys(y).map(Number).sort(function(a, b){ return b - a; });
}

  const v = sh.getRange(1, col, sh.getMaxRows(), 1).getValues();
  for (let i = v.length - 1; i >= 0; i--) if (String(v[i][0]).trim() !== '') return i + 1;

}
