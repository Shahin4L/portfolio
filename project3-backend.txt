function getTeamList() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const mailSheet = ss.getSheetByName(CONFIG.MAIL_SHEET);
    if (!mailSheet) return [];

    const lastRow = mailSheet.getLastRow();
    if (lastRow < 2) return [];

    const teamValues = mailSheet.getRange(2, 1, lastRow - 1, 1).getDisplayValues();
    const unique = {};

    teamValues.forEach(function(row) {
      const originalName = String(row[0] || "").trim();
      const key = normalizeTeamKey(originalName);

      if (!originalName || isSpecialAccessRow(key)) return;

      if (!unique[key]) {
        unique[key] = originalName;
      }
    });

    return Object.keys(unique)
      .map(function(key) { return unique[key]; })
      .sort(sortTeamNames);

  } catch (err) {
    return [];
  }
}

function getAllDashboardTeams() {
  return getTeamList();
}

function checkUserAccess(userEmail) {
  try {
    const cleanEmail = normalizeEmail(userEmail);

    if (!cleanEmail) {
      return {
        access: false,
        message: "ইমেইল প্রদান করা হয়নি।"
      };
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const mailSheet = ss.getSheetByName(CONFIG.MAIL_SHEET);

    if (!mailSheet) {
      return {
        access: false,
        message: "Mail Address sheet পাওয়া যায়নি।"
      };
    }

    const lastRow = mailSheet.getLastRow();
    const lastCol = Math.max(mailSheet.getLastColumn(), 1);

    if (lastRow < 2 || lastCol < 2) {
      return {
        access: false,
        message: "Mail Address শিটে কোনো ডেটা নেই।"
      };
    }

    const data = mailSheet
      .getRange(2, 1, lastRow - 1, lastCol)
      .getDisplayValues();

    const allowedTeamMap = {};
    let isSuperAdmin = false;

    for (let i = 0; i < data.length; i++) {
      const rowTeam = String(data[i][0] || "").trim();
      const teamKey = normalizeTeamKey(rowTeam);

      if (!rowTeam) continue;

      for (let col = 1; col < data[i].length; col++) {
        const emailCell = normalizeEmail(data[i][col]);

        if (!emailCell || emailCell !== cleanEmail) continue;

        if (teamKey === "super admin") {
          isSuperAdmin = true;
          continue;
        }

        if (isSpecialAccessRow(teamKey)) continue;

        if (rowTeam) {
          allowedTeamMap[teamKey] = rowTeam;
        }
      }
    }

    const allowedTeams = Object.keys(allowedTeamMap)
      .map(function(key) { return allowedTeamMap[key]; })
      .sort(sortTeamNames);

    if (!isSuperAdmin && allowedTeams.length === 0) {
      return {
        access: false,
        message: "⚠️ আপনার এই ইমেইলটি (" + cleanEmail + ") Mail Address লিস্টে পাওয়া যায়নি।"
      };
    }

    const allTeams = getTeamList();

    return {
      access: true,
      isSuperAdmin: isSuperAdmin,
      teams: allowedTeams,
      availableTeams: isSuperAdmin ? allTeams : allowedTeams
    };

  } catch (err) {
    return {
      access: false,
      message: "ত্রুটি: " + err.message
    };
  }
}

/**
 * Enhanced Dashboard Data Retrieval
 * Added selectedTeam and dateFilter options for UI interactions.
 */
function getDashboardData(userEmail, selectedTeam, dateFilter) {
  const auth = checkUserAccess(userEmail);

  if (!auth.access) {
    return {
      success: false,
      message: auth.message
    };
  }

  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const dataSheet = ss.getSheetByName(CONFIG.DATA_SHEET);

    if (!dataSheet) {
      return {
        success: false,
        message: "Data sheet পাওয়া যায়নি।"
      };
    }

    const lastRow = dataSheet.getLastRow();

    const emptyStats = { total: 0, pending: 0, inProgress: 0, resolved: 0, rejected: 0 };
    const emptyPercentages = { pending: 0, inProgress: 0, resolved: 0, rejected: 0 };

    if (lastRow < 2) {
      return {
        success: true,
        isSuperAdmin: auth.isSuperAdmin,
        allowedTeams: auth.teams || [],
        availableTeams: auth.availableTeams || [],
        stats: emptyStats,
        percentages: emptyPercentages,
        data: []
      };
    }

    const rows = dataSheet.getRange(2, 1, lastRow - 1, 12).getDisplayValues();
    const filteredData = [];
    const stats = { total: 0, pending: 0, inProgress: 0, resolved: 0, rejected: 0 };

    const allowedTeamKeys = {};
    (auth.teams || []).forEach(function(team) {
      allowedTeamKeys[normalizeTeamKey(team)] = true;
    });

    const targetTeamKey = selectedTeam ? normalizeTeamKey(selectedTeam) : null;

    rows.forEach(function(row, index) {
      const team = String(row[3] || "").trim();
      const teamKey = normalizeTeamKey(team);

      // Super Admin access check or regular Team user access check
      const canView = auth.isSuperAdmin || !!allowedTeamKeys[teamKey];
      if (!canView || !team) return;

      // Filter by dynamic selected team on Dashboard dropdown
      if (targetTeamKey && targetTeamKey !== "all" && teamKey !== targetTeamKey) {
        return;
      }

      const status = normalizeStatus(row[10]);

      stats.total++;
      if (status === "Pending") stats.pending++;
      else if (status === "In Progress") stats.inProgress++;
      else if (status === "Resolved") stats.resolved++;
      else if (status === "Rejected") stats.rejected++;

      filteredData.push({
        rowIdx: index + 2,
        date: row[0],
        name: row[1],
        mobile: row[2],
        team: team,
        category: row[4],
        problem: row[5],
        location: row[6],
        evidence: row[7],
        id: row[9],
        status: status,
        remarks: row[11] || ""
      });
    });

    filteredData.reverse();

    const percentages = {
      pending: stats.total ? Number(((stats.pending / stats.total) * 100).toFixed(1)) : 0,
      inProgress: stats.total ? Number(((stats.inProgress / stats.total) * 100).toFixed(1)) : 0,
      resolved: stats.total ? Number(((stats.resolved / stats.total) * 100).toFixed(1)) : 0,
      rejected: stats.total ? Number(((stats.rejected / stats.total) * 100).toFixed(1)) : 0
    };

    return {
      success: true,
      isSuperAdmin: auth.isSuperAdmin,
      allowedTeams: auth.teams || [],
      availableTeams: auth.availableTeams || [],
      stats: stats,
      percentages: percentages,
      data: filteredData
    };

  } catch (err) {
    return {
      success: false,
      message: "ত্রুটি: " + err.message
    };
  }
}

function updateDashboardStatus(userEmail, complaintId, newStatus, userNote, rowNumber) {
  try {
    const auth = checkUserAccess(userEmail);

    if (!auth.access) {
      return { success: false, message: auth.message };
    }

    const allowedStatuses = ["Pending", "In Progress", "Resolved", "Rejected"];
    const status = String(newStatus || "").trim();

    if (allowedStatuses.indexOf(status) === -1) {
      return { success: false, message: "অবৈধ Status নির্বাচন করা হয়েছে।" };
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const dataSheet = ss.getSheetByName(CONFIG.DATA_SHEET);

    if (!dataSheet) {
      return { success: false, message: "Data sheet পাওয়া যায়নি।" };
    }

    const lastRow = dataSheet.getLastRow();
    if (lastRow < 2) {
      return { success: false, message: "কোনো অভিযোগ পাওয়া যায়নি।" };
    }

    const rows = dataSheet.getRange(2, 1, lastRow - 1, 12).getDisplayValues();
    const searchId = String(complaintId || "").trim().toUpperCase();
    const requestedRow = Number(rowNumber || 0);

    // Complaint ID থাকলে ID দিয়ে খুঁজব। ID না থাকলে Dashboard থেকে পাঠানো
    // actual sheet row number দিয়ে update করব। তাই পুরনো/blank-ID complaint-ও update করা যাবে।
    if (!searchId && (!requestedRow || requestedRow < 2 || requestedRow > lastRow)) {
      return { success: false, message: "Complaint-এর valid row পাওয়া যায়নি।" };
    }

    const allowedTeamKeys = {};
    (auth.teams || []).forEach(function(team) {
      allowedTeamKeys[normalizeTeamKey(team)] = true;
    });

    for (let i = 0; i < rows.length; i++) {
      const sheetRow = i + 2;
      const rowId = String(rows[i][9] || "").trim().toUpperCase();
      const rowMatchesId = !!searchId && rowId === searchId;
      const rowMatchesNumber = !searchId && requestedRow === sheetRow;

      if (!rowMatchesId && !rowMatchesNumber) continue;

      const rawCurrentStatus = String(rows[i][10] || "").trim();
      const currentStatus = normalizeStatus(rawCurrentStatus);
      const currentStatusKey = rawCurrentStatus.toLowerCase();
      const complaintTeam = normalizeTeamKey(rows[i][3]);

      // Resolved / Rejected complaint server-side থেকে permanently locked থাকবে।
      // Status-এর case/extra-space variation থাকলেও lock কাজ করবে।
      if (
        currentStatusKey === "resolved" ||
        currentStatusKey === "rejected" ||
        currentStatus === "Resolved" ||
        currentStatus === "Rejected"
      ) {
        const lockedStatus =
          currentStatusKey === "rejected" || currentStatus === "Rejected"
            ? "Rejected"
            : "Resolved";

        return {
          success: false,
          locked: true,
          message: "এই Complaint ইতোমধ্যে " + lockedStatus + " হয়েছে। এখন আর Status পরিবর্তন করা যাবে না।"
        };
      }

      if (!auth.isSuperAdmin && !allowedTeamKeys[complaintTeam]) {
        return {
          success: false,
          message: "⚠️ এই অভিযোগটি আপনার Team-এর জন্য অনুমোদিত নয়।"
        };
      }

      dataSheet.getRange(i + 2, 11).setValue(status);
      dataSheet.getRange(i + 2, 12).setValue(String(userNote || "").trim());

      return {
        success: true,
        message: "Status সফলভাবে আপডেট হয়েছে।"
      };
    }

    return { success: false, message: "Complaint ID পাওয়া যায়নি।" };

  } catch (err) {
    return { success: false, message: "ত্রুটি: " + err.message };
  }
}

function normalizeEmail(value) {
  return String(value || "")
    .replace(/[\u200B-\u200D\uFEFF]/g, "")
    .replace(/\u00A0/g, " ")
    .trim()
    .toLowerCase();
}

function normalizeTeamKey(value) {
  return String(value || "")
    .trim()
    .replace(/\s+/g, " ")
    .toLowerCase();
}

function isSpecialAccessRow(teamKey) {
  return (
    teamKey === "all" ||
    teamKey === "admin" ||
    teamKey === "super admin"
  );
}

function sortTeamNames(a, b) {
  return String(a || "").localeCompare(
    String(b || ""),
    undefined,
    { numeric: true, sensitivity: "base" }
  );
}

function normalizeStatus(value) {
  const status = String(value || "").trim();

  if (
    status === "Pending" ||
    status === "In Progress" ||
    status === "Resolved" ||
    status === "Rejected"
  ) {
    return status;
  }

  return "Pending";
}


/**
 * =====================================================
 * GOOGLE FORM SUBMIT TRIGGER
 * =====================================================
 * Google Form থেকে নতুন response আসলে:
 * 1) Complaint ID তৈরি করবে
 * 2) Data sheet-এর J column-এ ID লিখবে
 * 3) K column-এ Pending সেট করবে
 * 4) Team অনুযায়ী email পাঠাবে
 *
 * IMPORTANT:
 * Apps Script > Triggers থেকে এই function-এর
 * "From spreadsheet" + "On form submit" trigger তৈরি করতে হবে।
 */
function onGoogleFormSubmit(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);

  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const dataSheet = ss.getSheetByName(CONFIG.DATA_SHEET);
    const mailSheet = ss.getSheetByName(CONFIG.MAIL_SHEET);

    if (!dataSheet) throw new Error("Data sheet পাওয়া যায়নি।");
    if (!mailSheet) throw new Error("Mail Address sheet পাওয়া যায়নি।");

    // Form submit trigger-এর event range থেকে row বের করা
    const targetRow = (e && e.range)
      ? e.range.getRow()
      : dataSheet.getLastRow();

    if (targetRow < 2) return;

    // A:H থেকে submitted data নেওয়া
    const rowValues = dataSheet
      .getRange(targetRow, 1, 1, 8)
      .getValues()[0];

    const rawDate = rowValues[0];
    const formattedDateTime = rawDate
      ? Utilities.formatDate(new Date(rawDate), CONFIG.TIMEZONE, "dd-MMM-yyyy hh:mm a")
      : Utilities.formatDate(new Date(), CONFIG.TIMEZONE, "dd-MMM-yyyy hh:mm a");

    const name = String(rowValues[1] || "").trim();
    const mobile = String(rowValues[2] || "").trim();
    const team = normalizeText(rowValues[3]);
    const category = String(rowValues[4] || "").trim();
    const problem = String(rowValues[5] || "").trim();
    const location = String(rowValues[6] || "").trim();
    const evidenceUrl = String(rowValues[7] || "").trim();

    if (!team) {
      Logger.log("Google Form row " + targetRow + ": Team পাওয়া যায়নি।");
      return;
    }

    // যদি আগে থেকেই Complaint ID থাকে, duplicate ID তৈরি করবে না
    const existingId = String(
      dataSheet.getRange(targetRow, 10).getDisplayValue() || ""
    ).trim();

    let complaintId = existingId;

    if (!complaintId) {
      complaintId =
        "CB-CMP-" +
        Utilities.formatDate(new Date(), CONFIG.TIMEZONE, "yyyyMMdd") +
        "-" +
        String(targetRow - 1).padStart(3, "0");
    }

    // J = Complaint ID, K = Status, M = Follow-up flag
    dataSheet.getRange(targetRow, 10).setNumberFormat("@");
    dataSheet.getRange(targetRow, 10).setValue(complaintId);
    dataSheet.getRange(targetRow, 11).setValue("Pending");

    // যদি M column থাকে, follow-up flag reset
    if (dataSheet.getMaxColumns() >= 13) {
      dataSheet.getRange(targetRow, 13).setValue("No");
    }

    SpreadsheetApp.flush();

    // Team অনুযায়ী recipient বের করা
    const emails = getTeamEmails(mailSheet, team);

    let emailSent = false;

    if (emails.length > 0) {
      try {
        emailSent = sendComplaintEmail(
          emails,
          team,
          category,
          complaintId,
          formattedDateTime,
          name,
          mobile,
          location,
          problem,
          evidenceUrl
        );
      } catch (emailErr) {
        Logger.log(
          "Google Form email error | " +
          complaintId +
          " | " +
          emailErr.message
        );
      }
    } else {
      Logger.log(
        "Google Form: Team '" +
        team +
        "' এর জন্য কোনো valid email পাওয়া যায়নি।"
      );
    }

    Logger.log(
      "Google Form processed | Row: " +
      targetRow +
      " | Complaint ID: " +
      complaintId +
      " | Email Sent: " +
      emailSent
    );

  } catch (err) {
    Logger.log(
      "Google Form Submit Error: " +
      err.message +
      "\n" +
      (err.stack || "")
    );
    throw err;

  } finally {
    lock.releaseLock();
  }
}


function submitComplaint(formData) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);

  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const dataSheet = ss.getSheetByName(CONFIG.DATA_SHEET);
    const mailSheet = ss.getSheetByName(CONFIG.MAIL_SHEET);

    if (!dataSheet) throw new Error('Data sheet পাওয়া যায়নি।');
    if (!mailSheet) throw new Error('Mail Address sheet পাওয়া যায়নি।');

    const name = String(formData.name || "").trim();
    const mobile = String(formData.mobile || "").trim();
    const team = normalizeText(formData.team);
    const category = String(formData.category || "").trim();
    const problem = String(formData.problem || "").trim();
    const confirmed = formData.confirmed === true;

    if (!name || !mobile || !team || !category || !problem || !confirmed) {
      throw new Error("সবগুলো প্রয়োজনীয় ঘর পূরণ করুন।");
    }

    const emails = getTeamEmails(mailSheet, team);
    const now = new Date();
    const formattedDateTime = Utilities.formatDate(now, CONFIG.TIMEZONE, "dd-MMM-yyyy hh:mm a");

    // Lock থাকার কারণে একই সময়ে ID duplicate হওয়ার ঝুঁকি কম থাকে।
    const nextRow = Math.max(dataSheet.getLastRow() + 1, 2);
    const complaintId = "CB-CMP-" + Utilities.formatDate(now, CONFIG.TIMEZONE, "yyyyMMdd") + "-" + String(nextRow - 1).padStart(3, "0");

    let evidenceUrl = "";
    if (formData.file && formData.file.data) {
      const fileSizeMB = formData.file.data.length * 0.75 / 1024 / 1024;
      if (fileSizeMB > CONFIG.MAX_FILE_SIZE_MB) {
        throw new Error("File size সর্বোচ্চ " + CONFIG.MAX_FILE_SIZE_MB + " MB হতে পারবে।");
      }

      const bytes = Utilities.base64Decode(formData.file.data);
      const blob = Utilities.newBlob(
        bytes,
        formData.file.type || "application/octet-stream",
        formData.file.name || ("Evidence-" + complaintId)
      );
      const file = DriveApp.createFile(blob);
      evidenceUrl = file.getUrl();
    }

    // Complaint ID-সহ পুরো row একবারে লিখছি।
    // Column J (Complaint ID) text হিসেবে রাখা হচ্ছে।
    dataSheet.getRange(nextRow, 10).setNumberFormat('@');
    dataSheet.getRange(nextRow, 1, 1, 12).setValues([[
      formattedDateTime,
      name,
      mobile,
      team,
      category,
      problem,
      formData.location || "N/A",
      evidenceUrl,
      "Yes",
      complaintId,
      "Pending",
      ""
    ]]);

    SpreadsheetApp.flush();

    // Logging আগে নিশ্চিত করা হচ্ছে। Email fail করলেও Complaint ID হারাবে না।
    let emailSent = false;
    let emailMessage = "";

    if (emails.length === 0) {
      emailMessage = "এই Team-এর জন্য Mail Address শিটে কোনো valid email পাওয়া যায়নি।";
    } else {
      try {
        emailSent = sendComplaintEmail(
          emails,
          team,
          category,
          complaintId,
          formattedDateTime,
          name,
          mobile,
          formData.location,
          problem,
          evidenceUrl
        );

        if (!emailSent) {
          emailMessage = "Email পাঠানো যায়নি।";
        }
      } catch (emailErr) {
        emailSent = false;
        emailMessage = "Complaint save হয়েছে, কিন্তু Email পাঠানো যায়নি: " + emailErr.message;
        Logger.log("sendComplaintEmail error for " + complaintId + ": " + emailErr.stack);
      }
    }

    return {
      success: true,
      complaintId: complaintId,
      date: formattedDateTime,
      emailSent: emailSent,
      emailMessage: emailMessage
    };

  } finally {
    lock.releaseLock();
  }
}

function trackComplaint(searchQuery) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const dataSheet = ss.getSheetByName(CONFIG.DATA_SHEET);
    if (!dataSheet) return { success: false, message: "Data sheet পাওয়া যায়নি।" };

    const lastRow = dataSheet.getLastRow();
    if (lastRow < 2) return { success: false, message: "কোনো তথ্য পাওয়া যায়নি।" };

    const data = dataSheet.getRange(2, 1, lastRow - 1, 12).getValues();
    const cleanQuery = String(searchQuery || "").trim().toLowerCase();
    const digitsOnlyQuery = cleanQuery.replace(/[^0-9]/g, '');

    const results = [];

    for (let i = 0; i < data.length; i++) {
      const row = data[i];
      const mobile = String(row[2] || "").trim(); 
      const complaintId = String(row[9] || "").trim(); 
      const status = String(row[10] || "Pending").trim(); 
      const remarks = String(row[11] || "কোন মন্তব্য নেই").trim(); 

      const cleanMobile = mobile.replace(/[^0-9]/g, '');
      const isIdMatch = complaintId.toLowerCase() === cleanQuery;
      const isPhoneMatch = digitsOnlyQuery.length >= 10 && cleanMobile.endsWith(digitsOnlyQuery);

      if (isIdMatch || isPhoneMatch) {
        results.push({
          id: complaintId,
          status: status || "Pending",
          remarks: remarks || "কোন মন্তব্য নেই"
        });
      }
    }

    if (results.length > 0) {
      return { success: true, data: results };
    } else {
      return { success: false, message: "প্রদত্ত ID বা নম্বর দিয়ে কোনো তথ্য পাওয়া যায়নি।" };
    }
  } catch (err) {
    return { success: false, message: "ত্রুটি: " + err.message };
  }
}

function updateComplaintStatusInSheet(complaintId, newStatus, userNote) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const dataSheet = ss.getSheetByName(CONFIG.DATA_SHEET);
    if (!dataSheet) return { success: false, message: "Data sheet পাওয়া যায়নি।" };

    const lastRow = dataSheet.getLastRow();
    if (lastRow < 2) return { success: false, message: "কোনো তথ্য পাওয়া যায়নি।" };

    const searchId = String(complaintId || "").trim().toUpperCase();
    if (!searchId) return { success: false, message: "Complaint ID পাওয়া যায়নি।" };

    const ids = dataSheet.getRange(2, 10, lastRow - 1, 1).getDisplayValues();
    const statuses = dataSheet.getRange(2, 11, lastRow - 1, 1).getDisplayValues();

    for (let i = 0; i < ids.length; i++) {
      if (String(ids[i][0] || "").trim().toUpperCase() !== searchId) continue;

      const currentStatus = normalizeStatus(statuses[i][0]);
      if (currentStatus === "Resolved" || currentStatus === "Rejected") {
        return {
          success: false,
          message: "এই Complaint ইতোমধ্যে " + currentStatus + " হয়েছে। এখন আর Status পরিবর্তন করা যাবে না।"
        };
      }

      dataSheet.getRange(i + 2, 11).setValue(newStatus);
      dataSheet.getRange(i + 2, 12).setValue(String(userNote || "").trim());
      return { success: true };
    }

    return { success: false, message: "Complaint ID পাওয়া যায়নি।" };
  } catch (err) {
    return { success: false, message: "ত্রুটি: " + err.message };
  }
}

function getTeamEmails(mailSheet, team) {
  const lastMailRow = mailSheet.getLastRow();
  const lastMailCol = mailSheet.getLastColumn();
  if (lastMailRow < 2 || lastMailCol < 2) return [];

  const mailData = mailSheet.getRange(2, 1, lastMailRow - 1, lastMailCol).getDisplayValues();
  const targetTeamKey = normalizeTeamKey(team);
  const unique = {};

  for (let i = 0; i < mailData.length; i++) {
    const teamName = normalizeText(mailData[i][0]);
    if (normalizeTeamKey(teamName) !== targetTeamKey) continue;

    for (let col = 1; col < mailData[i].length; col++) {
      const email = normalizeEmail(mailData[i][col]);
      if (email && isValidEmail(email)) unique[email] = true;
    }
    break;
  }

  return Object.keys(unique);
}

function emailRow(label, value) {
  return `<tr>
    <td style="padding:10px; background:#f9fafb; border:1px solid #e5e7eb; width:35%; font-weight:bold;">${escapeHtml(label)}</td>
    <td style="padding:10px; border:1px solid #e5e7eb;">${escapeHtml(value || "-")}</td>
  </tr>`;
}

function normalizeText(value) { return String(value || "").trim().replace(/\s+/g, " "); }
function isValidEmail(email) { return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email); }
function escapeHtml(value) { return String(value || "").replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;"); }
function formatMultilineText(value) { return escapeHtml(value).replace(/\r?\n/g, "<br>"); }

    // Keep the existing Data sheet structure intact.
    dataSheet.getRange(nextRow, 10).setNumberFormat("@");
    dataSheet.getRange(nextRow, 1, 1, 12).setValues([[

      "Anonymous",
      "",

      category,
      problem,
      location,
      evidenceUrl,
      "Yes",
      complaintId,
      "Pending",
      ""
    ]]);

    SpreadsheetApp.flush();

    let emailMessage = "";

        );
      }
    }

    return {
      success: true,
      complaintId: complaintId,
      date: formattedDateTime,
      emailSent: emailSent,
      emailMessage: emailMessage
    };
  } finally {
    lock.releaseLock();
  }
}

const JOB_CONFIG = {
  CIRCULAR_SHEET: "Job_Circular",
  APPLICATION_SHEET: "Job_Applications",
  RECRUITMENT_EMAIL_SHEET: "Recruitment email",
  DRIVE_FOLDER_NAME: "CarryBee Job Applications",
  MAX_CV_SIZE_MB: 10,
  TIMEZONE: "Asia/Dhaka",
  ALLOWED_EXTENSIONS: ["pdf", "doc", "docx"]
};

function getJobCirculars() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName(JOB_CONFIG.CIRCULAR_SHEET);
    if (!sheet) return { success:false, message:"Job_Circular sheet পাওয়া যায়নি।", jobs:[], activeCount:0 };

    const lastRow = sheet.getLastRow();
    if (lastRow < 2) return { success:true, jobs:[], activeCount:0 };

    const rows = sheet.getRange(2,1,lastRow-1,6).getDisplayValues();
    const jobs = [];

    rows.forEach(function(row, i) {
      const title = String(row[0] || "").trim();
      const department = String(row[1] || "").trim();
      const description = String(row[2] || "").trim();
      const salary = String(row[3] || "").trim();
      const youtube = String(row[4] || "").trim();
      const status = String(row[5] || "").trim();

      if (!title || status.toLowerCase() !== "active") return;

      jobs.push({
        rowNumber: i + 2,
        title: title,
        department: department,
        description: description,
        salary: salary,
        youtube: JOB_normalizeYouTubeUrl(youtube)
      });
    });

    return { success:true, jobs:jobs, activeCount:jobs.length };
  } catch (err) {
    Logger.log("getJobCirculars: " + err.stack);
    return { success:false, message:"Job data load করতে সমস্যা হয়েছে।", jobs:[], activeCount:0 };
  }
}

function JOB_normalizeYouTubeUrl(url) {
  const value = String(url || "").trim();
  if (!value) return "";

  let id = "";
  let m = value.match(/[?&]v=([A-Za-z0-9_-]{6,})/);
  if (m) id = m[1];
  m = m || value.match(/youtu\.be\/([A-Za-z0-9_-]{6,})/);
  if (!id && m) id = m[1];
  m = value.match(/youtube\.com\/embed\/([A-Za-z0-9_-]{6,})/);
  if (!id && m) id = m[1];
  m = value.match(/youtube\.com\/shorts\/([A-Za-z0-9_-]{6,})/);
  if (!id && m) id = m[1];

  return id ? "https://www.youtube.com/embed/" + id : "";
}

function submitJobApplication(formData) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);

  try {
    const name = String(formData && formData.name || "").trim();
    const mobile = String(formData && formData.mobile || "").trim();
    const email = normalizeEmail(formData && formData.email);
    const jobTitle = String(formData && formData.jobTitle || "").trim();
    const department = String(formData && formData.department || "").trim();
    const file = formData && formData.file;

    if (!name || !mobile || !email || !jobTitle || !file || !file.data) {
      return { success:false, message:"সব প্রয়োজনীয় তথ্য পূরণ করুন এবং CV সংযুক্ত করুন।" };
    }
    if (!isValidEmail(email)) return { success:false, message:"একটি valid email address দিন।" };

    const circular = getJobCirculars();
    if (!circular.success) return { success:false, message:circular.message };

    const job = circular.jobs.find(function(x) {
      return x.title === jobTitle && x.department === department;
    });
    if (!job) return { success:false, message:"এই Job Opportunity বর্তমানে Active নেই। পেজটি refresh করে আবার চেষ্টা করুন।" };

    const fileName = String(file.name || "").trim();
    const extMatch = fileName.toLowerCase().match(/\.([a-z0-9]+)$/);
    const ext = extMatch ? extMatch[1] : "";
    if (JOB_CONFIG.ALLOWED_EXTENSIONS.indexOf(ext) === -1) {
      return { success:false, message:"CV শুধু PDF, DOC অথবা DOCX format-এ দেওয়া যাবে।" };
    }

    const base64 = String(file.data || "").replace(/^data:[^;]+;base64,/, "");
    const sizeMB = base64.length * 0.75 / 1024 / 1024;
    if (sizeMB > JOB_CONFIG.MAX_CV_SIZE_MB) {
      return { success:false, message:"CV file size সর্বোচ্চ " + JOB_CONFIG.MAX_CV_SIZE_MB + " MB হতে পারবে।" };
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const appSheet = ss.getSheetByName(JOB_CONFIG.APPLICATION_SHEET);
    if (!appSheet) return { success:false, message:"Job_Applications sheet পাওয়া যায়নি।" };

    const now = new Date();
    const nextRow = Math.max(appSheet.getLastRow() + 1, 2);
    const applicationId =
      "JOB-" +
      Utilities.formatDate(now, JOB_CONFIG.TIMEZONE, "yyyyMMdd") +
      "-" +
      String(nextRow - 1).padStart(3, "0");

    const bytes = Utilities.base64Decode(base64);
    const blob = Utilities.newBlob(
      bytes,
      String(file.type || JOB_getMimeType(ext)),
      applicationId + "_" + fileName.replace(/[\\/:*?"<>|#%{}[\]]/g, "_")
    );

    const folderIter = DriveApp.getFoldersByName(JOB_CONFIG.DRIVE_FOLDER_NAME);
    const folder = folderIter.hasNext() ? folderIter.next() : DriveApp.createFolder(JOB_CONFIG.DRIVE_FOLDER_NAME);
    const driveFile = folder.createFile(blob);
    const cvUrl = driveFile.getUrl();

    appSheet.getRange(nextRow,1,1,8).setValues([[
      Utilities.formatDate(now, JOB_CONFIG.TIMEZONE, "dd-MMM-yyyy hh:mm a"),
      applicationId,
      jobTitle,
      department,
      name,
      mobile,
      email,
      cvUrl
    ]]);
    SpreadsheetApp.flush();

    let emailSent = false;
    let emailMessage = "";
    try {
      emailSent = JOB_sendApplicationEmail({
        applicationId: applicationId,
        dateTime: Utilities.formatDate(now, JOB_CONFIG.TIMEZONE, "dd-MMM-yyyy hh:mm a"),
        jobTitle: jobTitle,
        department: department,
        name: name,
        mobile: mobile,
        email: email,
        cvUrl: cvUrl,
        blob: blob
      });
    } catch (mailErr) {
      emailMessage = "Application save হয়েছে, কিন্তু notification email পাঠানো যায়নি।";
      Logger.log("JOB email: " + mailErr.stack);
    }

    return {
      success:true,
      applicationId:applicationId,
      emailSent:emailSent,
      emailMessage:emailMessage
    };
  } catch (err) {
    Logger.log("submitJobApplication: " + err.stack);
    return { success:false, message:"Application submit করতে সমস্যা হয়েছে: " + err.message };
  } finally {
    lock.releaseLock();
  }
}

function JOB_getMimeType(ext) {
  if (ext === "pdf") return "application/pdf";
  if (ext === "doc") return "application/msword";
  if (ext === "docx") return "application/vnd.openxmlformats-officedocument.wordprocessingml.document";
  return "application/octet-stream";
}

function JOB_getRecruitmentEmailRecipients(department) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(JOB_CONFIG.RECRUITMENT_EMAIL_SHEET);

  if (!sheet) {
    throw new Error('"' + JOB_CONFIG.RECRUITMENT_EMAIL_SHEET + '" sheet পাওয়া যায়নি।');
  }

  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();

  if (lastRow < 2 || lastCol < 2) {
    throw new Error('"' + JOB_CONFIG.RECRUITMENT_EMAIL_SHEET + '" sheet-এ data পাওয়া যায়নি।');
  }

  const rows = sheet.getRange(1, 1, lastRow, lastCol).getDisplayValues();
  const targetDepartment = normalizeText(department).toLowerCase();
  const toEmails = [];
  const ccEmails = [];

  for (let r = 1; r < rows.length; r++) {
    const rowDepartment = normalizeText(rows[r][0]).toLowerCase();
    if (!rowDepartment || rowDepartment !== targetDepartment) continue;

    // Column B = Primary Email. Multiple To emails can be separated by comma, semicolon or new line.
    collectJobEmails_(rows[r][1], toEmails);

    // Column C onward = CC Email 1, CC Email 2, ...
    for (let c = 2; c < rows[r].length; c++) {
      collectJobEmails_(rows[r][c], ccEmails);
    }
    break;
  }

  const uniqueTo = uniqueValidJobEmails_(toEmails);
  const uniqueCc = uniqueValidJobEmails_(ccEmails).filter(function(email) {
    return uniqueTo.indexOf(email) === -1;
  });

  if (!uniqueTo.length) {
    throw new Error(
      '"' + department + '" Department-এর জন্য "' +
      JOB_CONFIG.RECRUITMENT_EMAIL_SHEET +
      '" sheet-এ কোনো valid Primary Email পাওয়া যায়নি।'
    );
  }

  return {
    to: uniqueTo,
    cc: uniqueCc
  };
}

function collectJobEmails_(value, targetArray) {
  String(value || "")
    .split(/[;,\n\r]+/)
    .map(function(email) { return normalizeEmail(email); })
    .filter(function(email) { return isValidEmail(email); })
    .forEach(function(email) {
      targetArray.push(email);
    });
}

function uniqueValidJobEmails_(emails) {
  const seen = {};
  const result = [];

  (emails || []).forEach(function(email) {
    const clean = normalizeEmail(email);
    if (!clean || !isValidEmail(clean) || seen[clean]) return;
    seen[clean] = true;
    result.push(clean);
  });

  return result;
}

function JOB_emailRow(label, value) {
  return '<tr><td style="padding:8px 0;color:#64748b;width:34%;vertical-align:top;">' +
    escapeHtml(label) + '</td><td style="padding:8px 0;color:#0f172a;font-weight:700;">' +
    escapeHtml(value) + '</td></tr>';
}


/**
 * =====================================================
 * NOTICE BOARD ADD-ON
 * Sheet: "Notice Board"
 * Columns: Date | Titel | Description | Download Link | Status
 * Only Active notices are returned to the public UI.
 * =====================================================
 */
function getNoticeBoardData() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Notice Board");

    if (!sheet) {
      return {
        success: false,
        message: 'Notice Board sheet পাওয়া যায়নি।'
      };
    }

    const lastRow = sheet.getLastRow();
    if (lastRow < 2) {
      return {
        success: true,
        notices: []
      };
    }

    // Header row বাদ দিয়ে A:E থেকে display values নেওয়া হচ্ছে।
    const values = sheet.getRange(2, 1, lastRow - 1, 5).getDisplayValues();
    const notices = [];

    values.forEach(function(row) {
      const date = String(row[0] || "").trim();
      const title = String(row[1] || "").trim();
      const description = String(row[2] || "").trim();
      const downloadLink = String(row[3] || "").trim();
      const status = String(row[4] || "").trim().toLowerCase();

      // Blank status = inactive/publicly hidden.
      if (status !== "active") return;

      // Completely blank rows ignore.
      if (!date && !title && !description && !downloadLink) return;

      notices.push({
        date: date,
        title: title,
        description: description,
        downloadLink: downloadLink,
        status: "Active"
      });
    });

    // Sheet-এর নিচের/সর্বশেষ notice আগে দেখানো হবে।
    notices.reverse();

    return {
      success: true,
      notices: notices,
      activeCount: notices.length
    };

  } catch (err) {
    return {
      success: false,
      message: "Notice Board data load error: " + err.message
    };
  }
}



/**
 * =====================================================
 * CONTACT US DIRECTORY
 * Source sheet: Contract Info
 * Required headers:
 * Department | Name | Designation | Photo | Mobile | WhatsApp |
 * Email | Calendar Connected | Status
 * Only Status = Active entries are returned.
 * =====================================================
 */
function getContactDirectory() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Contract Info");
    if (!sheet) return { success:false, message:"Contract Info sheet পাওয়া যায়নি।", contacts:[] };

    const lastRow = sheet.getLastRow();
    const lastCol = sheet.getLastColumn();
    if (lastRow < 2 || lastCol < 1) return { success:true, contacts:[], departments:[] };

    const values = sheet.getRange(1, 1, lastRow, lastCol).getDisplayValues();
    const headers = values[0].map(function(h){ return normalizeContactHeader(h); });
    const col = {};
    headers.forEach(function(h, i){ if (h && col[h] === undefined) col[h] = i; });

    const required = ["department","name","designation","photo","mobile","whatsapp","email","calendar connected","status"];
    const missing = required.filter(function(h){ return col[h] === undefined; });
    if (missing.length) {
      return { success:false, message:"Contract Info sheet-এর header পাওয়া যায়নি: " + missing.join(", "), contacts:[] };
    }

    const contacts = [];
    const departmentsMap = {};

    for (let r = 1; r < values.length; r++) {
      const row = values[r];
      const status = String(row[col.status] || "").trim();
      if (status.toLowerCase() !== "active") continue;

      const department = String(row[col.department] || "").trim();
      const name = String(row[col.name] || "").trim();
      if (!department || !name) continue;

      contacts.push({
        department: department,
        name: name,
        designation: String(row[col.designation] || "").trim(),
        photo: normalizeContactPhotoUrl(row[col.photo]),
        mobile: normalizeContactPhone(row[col.mobile]),
        whatsapp: normalizeContactWhatsApp(row[col.whatsapp]),
        email: String(row[col.email] || "").trim(),
        calendarConnected: String(row[col["calendar connected"]] || "").trim(),
        status: "Active"
      });

      const depKey = normalizeContactKey(department);
      if (!departmentsMap[depKey]) departmentsMap[depKey] = department;
    }

    contacts.sort(function(a,b){
      const d = String(a.department).localeCompare(String(b.department), undefined, {sensitivity:"base"});
      return d || String(a.name).localeCompare(String(b.name), undefined, {sensitivity:"base"});
    });

    const departments = Object.keys(departmentsMap)
      .map(function(k){ return departmentsMap[k]; })
      .sort(function(a,b){ return String(a).localeCompare(String(b), undefined, {sensitivity:"base"}); });

    return { success:true, contacts:contacts, departments:departments };
  } catch (err) {
    Logger.log("getContactDirectory: " + err.stack);
    return { success:false, message:"Contact data load করতে সমস্যা হয়েছে।", contacts:[] };
  }
}

function normalizeContactHeader(value) {
  return String(value || "").replace(/[\u200B-\u200D\uFEFF]/g, "").replace(/\u00A0/g, " ").trim().toLowerCase().replace(/\s+/g, " ");
}
function normalizeContactKey(value) {
  return String(value || "").trim().replace(/\s+/g, " ").toLowerCase();
}
function normalizeContactPhone(value) {
  return String(value || "").trim().replace(/[^0-9+]/g, "");
}
function normalizeContactWhatsApp(value) {
  return String(value || "")
    .replace(/[\u200B-\u200D\uFEFF]/g, "")
    .replace(/\u00A0/g, " ")
    .trim();
}
function normalizeContactPhotoUrl(value) {
  const url = String(value || "").trim();
  if (!url) return "";
  let m = url.match(/\/d\/([A-Za-z0-9_-]{10,})/);
  if (!m) m = url.match(/[?&]id=([A-Za-z0-9_-]{10,})/);
  if (m && m[1]) return "https://drive.google.com/thumbnail?id=" + encodeURIComponent(m[1]) + "&sz=w400-h400";
  return url;
}

/* =========================================================
 * ORGANOGRAM ADD-ON ONLY
 * Existing complaint / mail / dashboard / contact logic is untouched.
 * Sheet: Organogram
 * Headers: Division | Name | Designation | Team | Reports To | Photo
 * ========================================================= */
function getOrganogramData() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Organogram");
    if (!sheet) {
      return {
        success: false,
        message: '"Organogram" নামে Sheet পাওয়া যায়নি।',
        rows: [],
        divisions: []
      };
    }

    const lastRow = sheet.getLastRow();
    const lastCol = sheet.getLastColumn();
    if (lastRow < 2 || lastCol < 1) {
      return { success: true, rows: [], divisions: [] };
    }

    const values = sheet.getRange(1, 1, lastRow, Math.max(lastCol, 6)).getDisplayValues();
    const headers = values[0].map(function(h) {
      return String(h || "").replace(/[\u200B-\u200D\uFEFF]/g, "").replace(/\u00A0/g, " ").trim().toLowerCase().replace(/\s+/g, " ");
    });

    const col = {};
    headers.forEach(function(h, i) {
      if (h && col[h] === undefined) col[h] = i;
    });

    const required = ["division", "name", "designation", "team", "reports to", "photo"];
    const missing = required.filter(function(h) { return col[h] === undefined; });
    if (missing.length) {
      return {
        success: false,
        message: "Organogram Sheet-এর header পাওয়া যায়নি: " + missing.join(", "),
        rows: [],
        divisions: []
      };
    }

    const rows = [];
    const divisionsMap = {};

    for (let r = 1; r < values.length; r++) {
      const row = values[r];
      const name = String(row[col["name"]] || "").trim();
      if (!name) continue;

      const division = String(row[col["division"]] || "").trim();
      const designation = String(row[col["designation"]] || "").trim();
      const team = String(row[col["team"]] || "").trim();
      const reportsTo = String(row[col["reports to"]] || "").trim();
      const photo = String(row[col["photo"]] || "").trim();

      rows.push({
        division: division,
        name: name,
        designation: designation,
        team: team,
        reportsTo: reportsTo,
        photo: photo
      });

      if (division) {
        const key = division.toLowerCase().replace(/\s+/g, " ");
        if (!divisionsMap[key]) divisionsMap[key] = division;
      }
    }

    const divisions = Object.keys(divisionsMap).map(function(k) { return divisionsMap[k]; });
    divisions.sort(function(a, b) {
      return String(a).localeCompare(String(b), undefined, { numeric: true, sensitivity: "base" });
    });

    return { success: true, rows: rows, divisions: divisions };
  } catch (err) {
    Logger.log("getOrganogramData: " + err.stack);
    return {
      success: false,
      message: "Organogram data load করতে সমস্যা হয়েছে।",
      rows: [],
      divisions: []
    };
  }
}
