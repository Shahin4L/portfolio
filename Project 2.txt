const CFG = {
  TZ: 'Asia/Dhaka',
  EMP: 'Employees',
  OBS: 'PTN Observations',
  USR: 'Users',
  TITLE: 'Central Sort Productivity Monitor',
  MIN_SEC: 60,
  OTP_TTL: 600,                    // OTP valid for 10 minutes
  DASH_ROLES: ['admin', 'manager'] // roles that can see Dashboard/Analytics
};

const OBS_HEADERS = ['Observation ID', 'Observation Date', 'Observer', 'Employee ID', 'Employee Name',
  'Department', 'Start Time', 'End Time', 'Duration (Minutes)', 'PTN Count', 'PTNs per Hour',
  'Remarks', 'Paused (Minutes)', 'Process', 'Target per Hour'];
const USER_HEADERS = ['User_Email', 'Full_Name', 'Role', 'Department', 'Status'];

function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle(CFG.TITLE)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

function ss_() {
  const s = SpreadsheetApp.getActiveSpreadsheet();
  if (!s) throw new Error('No active spreadsheet. Open Apps Script from the tracker spreadsheet.');
  return s;
}

/* Run once from the editor (also authorizes Mail permission). */
function setupTracker() {
  const ss = ss_();
  [[CFG.OBS, OBS_HEADERS], [CFG.USR, USER_HEADERS]].forEach(([name, h]) => {
    let sh = ss.getSheetByName(name) || ss.insertSheet(name);
    sh.getRange(1, 1, 1, h.length).setValues([h]).setFontWeight('bold');
    sh.setFrozenRows(1);
  });
  // Process list lives in Users!J:L (Item | Per Mint Productivity | Per Hour Productivity)
  const u = ss.getSheetByName(CFG.USR);
  u.getRange(1, 10, 1, 3).setValues([['Item', 'Per Mint Productivity', 'Per Hour Productivity']]).setFontWeight('bold');
  if (u.getLastRow() < 2 || !u.getRange('J2').getValue()) {
    u.getRange('J2:L3').setValues([['Inbound Process', 1, 60], ['PTN Process', 2, 120]]); // sample - edit freely
  }
  const o = ss.getSheetByName(CFG.OBS);
  o.getRange('O:O').setNumberFormat('0.00');
  o.getRange('B:B').setNumberFormat('dd-mmm-yyyy hh:mm:ss');
  o.getRange('G:H').setNumberFormat('dd-mmm-yyyy hh:mm:ss');
  o.getRange('I:I').setNumberFormat('0.0000');
  o.getRange('K:K').setNumberFormat('0.00');
  o.getRange('M:M').setNumberFormat('0.00');
  MailApp.getRemainingDailyQuota();
  return { success: true };
}

/* ---------------- AUTH ---------------- */

function findUser_(email) {
  const sh = ss_().getSheetByName(CFG.USR);
  if (!sh) throw new Error('Users sheet not found. Run setupTracker first.');
  const n = sh.getLastRow();
  if (n < 2) return null;
  const rows = sh.getRange(2, 1, n - 1, 5).getDisplayValues();
  for (const r of rows) {
    if (r[0].trim().toLowerCase() === email && r[4].trim().toLowerCase() === 'active') {
      return { email: email, name: r[1].trim(), role: r[2].trim().toLowerCase(), dept: r[3].trim() };
    }
  }
  return null;
}

function hash_(s) {
  return Utilities.base64Encode(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, s));
}

function requestOtp(email) {
  email = String(email || '').trim().toLowerCase();
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) throw new Error('Enter a valid email address.');
  if (!findUser_(email)) throw new Error('This email is not authorized or is inactive.');

  const cache = CacheService.getScriptCache();
  if (cache.get('cool_' + email)) throw new Error('Please wait a minute before requesting a new code.');

  const code = String(parseInt(Utilities.getUuid().replace(/-/g, '').slice(0, 8), 16) % 900000 + 100000);
  cache.put('otp_' + email, JSON.stringify({ h: hash_(email + code), a: 0 }), CFG.OTP_TTL);
  cache.put('cool_' + email, '1', 60);

  MailApp.sendEmail({
    to: email,
    subject: CFG.TITLE + ' - your verification code',
    htmlBody: '<div style="font-family:Arial,sans-serif;max-width:420px;margin:auto;padding:24px;border:1px solid #e5e7eb;border-radius:12px">' +
      '<h2 style="margin:0 0 8px;color:#4f46e5">' + CFG.TITLE + '</h2>' +
      '<p style="color:#475569">Use this code to sign in:</p>' +
      '<div style="font-size:34px;font-weight:700;letter-spacing:8px;margin:14px 0">' + code + '</div>' +
      '<p style="color:#94a3b8;font-size:12px">Valid for 10 minutes. If you did not request this, ignore this email.</p></div>'
  });
  return { ok: true };
}

function verifyOtp(email, code) {
  email = String(email || '').trim().toLowerCase();
  code = String(code || '').trim();
  const cache = CacheService.getScriptCache();
  const raw = cache.get('otp_' + email);
  if (!raw) throw new Error('Code expired. Please request a new one.');
  const o = JSON.parse(raw);
  if (o.a >= 5) { cache.remove('otp_' + email); throw new Error('Too many attempts. Request a new code.'); }
  if (o.h !== hash_(email + code)) {
    o.a++;
    cache.put('otp_' + email, JSON.stringify(o), CFG.OTP_TTL);
    throw new Error('Incorrect code.');
  }
  cache.remove('otp_' + email);
  const user = findUser_(email);
  if (!user) throw new Error('This account is inactive.');

  // Session stays valid until the user logs out manually.
  const token = Utilities.getUuid() + Utilities.getUuid();
  PropertiesService.getScriptProperties()
    .setProperty('S_' + token, JSON.stringify({ e: email, t: Date.now() }));
  return Object.assign({ token: token }, appData_(user));
}

function auth_(token) {
  if (!token) throw new Error('SESSION_EXPIRED');
  const raw = PropertiesService.getScriptProperties().getProperty('S_' + token);
  if (!raw) throw new Error('SESSION_EXPIRED');
  const user = findUser_(JSON.parse(raw).e);
  if (!user) { logout(token); throw new Error('SESSION_EXPIRED'); }
  return user;
}

function logout(token) {
  if (token) PropertiesService.getScriptProperties().deleteProperty('S_' + token);
  return true;
}

function appData_(user) {
  return {
    title: CFG.TITLE,
    user: user,
    canDash: CFG.DASH_ROLES.indexOf(user.role) > -1,
    processes: getProcesses_(),
    employees: getActiveEmployees_()
  };
}

function getAppData(token) {
  return appData_(auth_(token));
}

/* ---------------- DATA ---------------- */

// Employees: A=ID, B=Name, C=Status, D=Division, E=Department
function getActiveEmployees_() {
  const sh = ss_().getSheetByName(CFG.EMP);
  if (!sh) throw new Error('Employees sheet was not found.');
  const n = sh.getLastRow();
  if (n < 2) return [];
  return sh.getRange(2, 1, n - 1, 5).getDisplayValues()
    .filter(r => r[0].trim() && r[1].trim() && r[2].trim().toLowerCase() === 'active')
    .map(r => ({ id: r[0].trim(), name: r[1].trim(), division: r[3].trim(), department: r[4].trim() }))
    .sort((a, b) => a.name.localeCompare(b.name));
}

// Users sheet: J = Item, K = Per Minute, L = Per Hour
function getProcesses_() {
  const sh = ss_().getSheetByName(CFG.USR);
  if (!sh || sh.getMaxColumns() < 12 || sh.getLastRow() < 2) return [];
  const num = v => { const n = parseFloat(String(v).replace(/,/g, '')); return isNaN(n) ? 0 : n; };
  const seen = {}, out = [];
  sh.getRange(2, 10, sh.getLastRow() - 1, 3).getDisplayValues().forEach(r => {
    const name = r[0].trim();
    if (!name || seen[name.toLowerCase()]) return;
    seen[name.toLowerCase()] = 1;
    let pm = num(r[1]), ph = num(r[2]);
    if (!ph && pm) ph = pm * 60;
    if (!pm && ph) pm = ph / 60;
    out.push({ name: name, perMin: pm, perHr: ph });
  });
  return out;
}

// Compact rows: s=start ms, e=end ms, o=observer, id, nm, d=dept, m=minutes, p=units, pr=process, tg=target/hr
function getObservations(token) {
  const user = auth_(token);
  const sh = ss_().getSheetByName(CFG.OBS);
  if (!sh || sh.getLastRow() < 2) return [];
  const all = CFG.DASH_ROLES.indexOf(user.role) > -1;
  const vals = sh.getRange(2, 1, sh.getLastRow() - 1, Math.min(15, sh.getMaxColumns())).getValues();
  const out = [];
  vals.forEach(r => {
    const s = r[6] instanceof Date ? r[6].getTime() : NaN;
    const e = r[7] instanceof Date ? r[7].getTime() : NaN;
    const m = Number(r[8]), p = Number(r[9]);
    if (isNaN(s) || !(m > 0) || isNaN(p)) return;
    if (!all && String(r[2]).trim() !== user.name) return;
    out.push({ s: s, e: e, o: String(r[2]), id: String(r[3]), nm: String(r[4]), d: String(r[5]), m: m, p: p,
      pr: String(r[13] || '').trim() || 'Unassigned', tg: Number(r[14]) || 0 });
  });
  return out;
}

function submitObservation(token, payload) {
  const user = auth_(token);
  if (!payload || typeof payload !== 'object') throw new Error('Invalid submission.');

  const employeeId = String(payload.employeeId || '').trim();
  const remarks = String(payload.remarks || '').trim();
  const ptn = Number(payload.ptnCount);
  const startMs = Number(payload.startTimeMs);
  const endMs = Number(payload.endTimeMs);
  const durMs = Number(payload.durationMs);

  if (!employeeId) throw new Error('Please select an employee.');
  if (payload.ptnCount === '' || !Number.isInteger(ptn) || ptn < 0)
    throw new Error('PTN Count must be a whole number, 0 or above.');
  if (!(startMs > 0) || !(endMs > startMs) || !(durMs > 0) || durMs > (endMs - startMs) + 1500)
    throw new Error('Invalid observation time data.');
  if (durMs / 1000 < CFG.MIN_SEC)
    throw new Error('Observation must be at least ' + CFG.MIN_SEC + ' seconds of active time.');

  const proc = getProcesses_().find(x => x.name === String(payload.process || '').trim());
  if (!proc) throw new Error('Please select a valid process.');
  const emp = getActiveEmployees_().find(x => x.id === employeeId);
  if (!emp) throw new Error('Employee not found or not Active.');

  const durMin = durMs / 60000;
  const pausedMin = Math.max(0, (endMs - startMs - durMs) / 60000);
  const pph = ptn / (durMin / 60);

  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const ss = ss_();
    let sh = ss.getSheetByName(CFG.OBS);
    if (!sh) { setupTracker(); sh = ss.getSheetByName(CFG.OBS); }
    const id = Utilities.getUuid();
    sh.appendRow([id, new Date(), user.name, emp.id, emp.name, emp.department,
      new Date(startMs), new Date(endMs), durMin, ptn, pph, remarks, pausedMin, proc.name, proc.perHr]);
    const r = sh.getLastRow();
    sh.getRange(r, 2).setNumberFormat('dd-mmm-yyyy hh:mm:ss');
    sh.getRange(r, 7, 1, 2).setNumberFormat('dd-mmm-yyyy hh:mm:ss');
    sh.getRange(r, 9).setNumberFormat('0.0000');
    sh.getRange(r, 11).setNumberFormat('0.00');
    sh.getRange(r, 13).setNumberFormat('0.00');
    sh.getRange(r, 15).setNumberFormat('0.00');
    return { success: true, observationId: id, employeeName: emp.name,
      durationMinutes: durMin, ptnCount: ptn, ptnsPerHour: pph,
      process: proc.name, target: proc.perHr };
  } finally {
    lock.releaseLock();
  }
}
